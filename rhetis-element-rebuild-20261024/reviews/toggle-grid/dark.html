SELECT

hub_key,

SUM(box_qty) AS packed_qty,

AVG(lag_h) AS mean_lag

FROM cargo_day

WHERE day_key >= '2098-04-01'

AND hub_key != '보류'

GROUP BY hub_key

HAVING SUM(box_qty) > 140

ORDER BY mean_lag ASC;
