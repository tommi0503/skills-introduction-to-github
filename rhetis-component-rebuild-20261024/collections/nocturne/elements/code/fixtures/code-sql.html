-- 주별 접수 흐름 분석

WITH weekly AS (

SELECT week_id,

region_name,

COUNT(*) AS total_count

FROM case_log

WHERE status = 'closed'

GROUP BY week_id, region_name

)

SELECT week_id,

region_name,

total_count,

SUM(total_count) OVER (

PARTITION BY week_id

) AS weekly_count

FROM weekly

ORDER BY week_id,

total_count DESC;
